Code figure


export class CircuitBreaker {
  private failures = 0;
  private openedAt: number | null = null;
  constructor(private options: { threshold: number; cooldown: number }) {}
  async run<T>(task: () => Promise<T>): Promise<T> {
    if (this.openedAt && Date.now() - this.openedAt < this.options.cooldown)
      throw new PaymentError({ code: "breaker_open" });
    try {
      const result = await task();
      this.failures = 0;
      this.openedAt = null;
      return result;
    } catch (error) {
      if (++this.failures >= this.options.threshold) this.openedAt = Date.now();
      throw error;
    }
  }
}

Math


The expected added latency per request while the breaker is closed is \mathbb{E}[\Delta] = p_f \cdot t_{timeout}.


P(\text{open}) = 1 - \sum_{k=0}^{n-1} \binom{N}{k} p_f^k (1-p_f)^{N-k}
